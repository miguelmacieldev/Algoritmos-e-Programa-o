inicio
   retorne matDist[codOrigem, codDestino]
fimfuncao

funcao calcularCusto(distanciaKm: inteiro; pesoKg: real; prioEntrega: inteiro): real
var
   percPeso, percPrio: real
inicio
   se pesoKg <= PESO_LIMITE_1 entao
      percPeso <- 0
   senao
      se (pesoKg > PESO_LIMITE_1) e (pesoKg <= PESO_LIMITE_2) entao
         percPeso <- ACRESC_PESO_1
      senao
         se (pesoKg > PESO_LIMITE_2) e (pesoKg <= PESO_LIMITE_3) entao
            percPeso <- ACRESC_PESO_2
         senao
            percPeso <- ACRESC_PESO_3
         fimse
      fimse
   fimse

   percPrio <- 0

   se prioEntrega = URGENTE entao
      percPrio <- ACRESC_URGENTE
   fimse

   se prioEntrega = PRIORITARIA entao
      percPrio <- ACRESC_PRIORITARIA
   fimse

   retorne distanciaKm * VALOR_KM * (1 + percPeso + percPrio)
fimfuncao

funcao localizarEntrega(codigoBusca: inteiro): inteiro
var
   i, achou: inteiro
inicio
   achou <- 0
   i <- 1

   enquanto (i <= qtdEntregas) e (achou = 0) faca
      se vCodigo[i] = codigoBusca entao
         achou <- i
      fimse
      i <- i + 1
   fimenquanto

   retorne achou
fimfuncao

procedimento lerInteiroNoIntervalo(campo: caractere; minimo: inteiro; maximo: inteiro; var valorLido: inteiro)
inicio
   repita
      escreva(campo, " (", minimo, " a ", maximo, "): ")
      leia(valorLido)

      se (valorLido < minimo) ou (valorLido > maximo) entao
         escreval("ERRO: ", campo, " invalido(a)! Digite um valor de ", minimo, " a ", maximo, ".")
      fimse
   ate (valorLido >= minimo) e (valorLido <= maximo)
fimprocedimento

procedimento cadastrarEntrega
var
   codNovo, origNovo, destNovo, prioNova, indice: inteiro
   nomeNovo: caractere
   pesoNovo, valorNovo: real
inicio
   escreval("=== CADASTRAR ENTREGA ===")

   se qtdEntregas >= MAX_ENTREGAS entao
      escreval("ERRO: capacidade maxima de ", MAX_ENTREGAS, " entregas atingida.")
   senao
      escreval("Entregas cadastradas: ", qtdEntregas, " de ", MAX_ENTREGAS)

      repita
         escreva("Codigo da entrega (inteiro maior que 0): ")
         leia(codNovo)

         se codNovo <= 0 entao
            escreval("ERRO: o codigo deve ser maior que zero.")
         senao
            se localizarEntrega(codNovo) > 0 entao
               escreval("ERRO: codigo ", codNovo, " duplicado!")
               codNovo <- 0
            fimse
         fimse
      ate codNovo > 0

      repita
         escreva("Nome/descricao do destinatario: ")
         leia(nomeNovo)

         se compr(nomeNovo) = 0 entao
            escreval("ERRO: o destinatario nao pode ficar vazio.")
         fimse
      ate compr(nomeNovo) > 0

      exibirRegioes
      lerInteiroNoIntervalo("Regiao de origem", 1, MAX_REGIOES, origNovo)

      repita
         lerInteiroNoIntervalo("Regiao de destino", 1, MAX_REGIOES, destNovo)

         se destNovo = origNovo entao
            escreval("ERRO: origem e destino nao podem ser iguais.")
         fimse
      ate destNovo <> origNovo

      repita
         escreva("Peso da carga em kg (maior que 0 e ate ", PESO_MAXIMO:6:2, "): ")
         leia(pesoNovo)

         se (pesoNovo <= 0) ou (pesoNovo > PESO_MAXIMO) entao
            escreval("ERRO: peso invalido (zero, negativo ou acima da capacidade maxima).")
         fimse
      ate (pesoNovo > 0) e (pesoNovo <= PESO_MAXIMO)

      repita
         escreva("Valor da mercadoria em R$ (0 ou mais): ")
         leia(valorNovo)

         se valorNovo < 0 entao
            escreval("ERRO: o valor da mercadoria nao pode ser negativo.")
         fimse
      ate valorNovo >= 0

      escreval("Prioridades: 1 - Urgente | 2 - Prioritaria | 3 - Normal")
      lerInteiroNoIntervalo("Prioridade", URGENTE, NORMAL, prioNova)

      indice <- qtdEntregas + 1

      vCodigo[indice] <- codNovo
      vDestinatario[indice] <- nomeNovo
      vOrigem[indice] <- origNovo
      vDestino[indice] <- destNovo
      vPeso[indice] <- pesoNovo
      vValor[indice] <- valorNovo
      vPrioridade[indice] <- prioNova
      vDistancia[indice] <- obterDistancia(origNovo, destNovo)
      vStatus[indice] <- AGUARDANDO
      vCusto[indice] <- calcularCusto(vDistancia[indice], pesoNovo, prioNova)

      qtdEntregas <- indice

      escreval("Entrega cadastrada com sucesso!")
      exibirDetalhesEntrega(indice)
   fimse
fimprocedimento

funcao precede(idxA: inteiro; idxB: inteiro): logico
var
   resultado: logico
inicio
   resultado <- falso

   se (vPrioridade[idxA] < vPrioridade[idxB]) ou ((vPrioridade[idxA] = vPrioridade[idxB]) e (vValor[idxA] > vValor[idxB])) ou ((vPrioridade[idxA] = vPrioridade[idxB]) e (vValor[idxA] = vValor[idxB]) e (vPeso[idxA] > vPeso[idxB])) ou ((vPrioridade[idxA] = vPrioridade[idxB]) e (vValor[idxA] = vValor[idxB]) e (vPeso[idxA] = vPeso[idxB]) e (vCodigo[idxA] < vCodigo[idxB])) entao
      resultado <- verdadeiro
   fimse

   retorne resultado
fimfuncao

procedimento ordenarEntregas
var
   passada, j: inteiro
   trocou, deveTrocar: logico
inicio
   passada <- 1

   repita
      trocou <- falso

      para j de 1 ate qtdEntregas - passada faca
         deveTrocar <- precede(j + 1, j)

         se deveTrocar entao
            trocarEntregas(j, j + 1)
            trocou <- verdadeiro
         fimse
      fimpara

      passada <- passada + 1
   ate (nao trocou) ou (passada >= qtdEntregas)
fimprocedimento

procedimento exibirFilaDespacho

   i, posicaoFila, totalFila: inteiro
inicio
   escreval("=== FILA DE DESPACHO ===")

   se qtdEntregas = 0 entao
      escreval("Nenhuma entrega cadastrada.")
   senao

funcao transicaoValida(statusAtual: inteiro; statusNovo: inteiro): logico
var
   valida: logico
inicio
   valida <- falso

   se ((statusAtual = AGUARDANDO) e (statusNovo = EM_TRANSPORTE)) ou ((statusAtual = EM_TRANSPORTE) e (statusNovo = ENTREGUE)) entao
      valida <- verdadeiro
   fimse

   retorne valida
fimfuncao

procedimento mudarStatusEntrega(statusNovo: inteiro)

   codBusca, indice: inteiro
   permitida: logico

            se statusNovo = EM_TRANSPORTE entao

procedimento exibirMatrizEAnalise
var
   i, j: inteiro
   maiorDist, menorDist, somaDist, contDist: inteiro
   mediaDist: real
inicio
   escreval("=== MATRIZ DE DISTANCIAS (km) ===")

   para j de 1 ate MAX_REGIOES faca
      escreva(j, "=", nomeRegiao[j], "  ")
   fimpara

   escreval("")
   escreva("Orig/Dest")

   para j de 1 ate MAX_REGIOES faca
      escreva(j:6)
   fimpara

   escreval("")

   maiorDist <- 0
   menorDist <- 999999
   somaDist <- 0
   contDist <- 0

   para i de 1 ate MAX_REGIOES faca
      escreva(i:5, "    ")

      para j de 1 ate MAX_REGIOES faca
         escreva(matDist[i, j]:6)

         se matDist[i, j] > maiorDist entao
            maiorDist <- matDist[i, j]
         fimse

         se i <> j entao
            somaDist <- somaDist + matDist[i, j]
            contDist <- contDist + 1

            se (matDist[i, j] <> 0) e (matDist[i, j] < menorDist) entao
               menorDist <- matDist[i, j]
            fimse
         fimse
      fimpara

      escreval("")
   fimpara

   mediaDist <- somaDist / contDist

   escreval("")
   escreval("Maior distancia existente ........: ", maiorDist, " km")
   escreval("Menor distancia (diferente de 0) .: ", menorDist, " km")
   escreval("Media das distancias entre regioes: ", mediaDist:6:2, " km")
fimprocedimento

procedimento exibirEstatisticas
var
   i, j: inteiro
   idxMaiorValor, idxMaiorPeso, idxMaiorDist, maiorQtdRegiao: inteiro
   somaPeso, somaValor, somaCusto: real
   mediaPeso, mediaValor, mediaCusto: real
inicio
   escreval("=== ESTATISTICAS ===")

   se qtdEntregas = 0 entao
      escreval("Nenhuma entrega cadastrada. Nao ha estatisticas para exibir.")
   senao
      somaPeso <- 0
      somaValor <- 0
      somaCusto <- 0

      idxMaiorValor <- 1
      idxMaiorPeso <- 1
      idxMaiorDist <- 1

      para j de 1 ate QTD_NIVEIS faca
         contStatus[j] <- 0
         contPrio[j] <- 0
      fimpara

      para j de 1 ate MAX_REGIOES faca
         contRegiao[j] <- 0
      fimpara

      para i de 1 ate qtdEntregas faca
         contStatus[vStatus[i]] <- contStatus[vStatus[i]] + 1
         contPrio[vPrioridade[i]] <- contPrio[vPrioridade[i]] + 1
         contRegiao[vDestino[i]] <- contRegiao[vDestino[i]] + 1

         somaPeso <- somaPeso + vPeso[i]
         somaValor <- somaValor + vValor[i]
         somaCusto <- somaCusto + vCusto[i]

         se vValor[i] > vValor[idxMaiorValor] entao
            idxMaiorValor <- i
         fimse

         se vPeso[i] > vPeso[idxMaiorPeso] entao
            idxMaiorPeso <- i
         fimse

         se vDistancia[i] > vDistancia[idxMaiorDist] entao
            idxMaiorDist <- i
         fimse
      fimpara

      mediaPeso <- somaPeso / qtdEntregas
      mediaValor <- somaValor / qtdEntregas
      mediaCusto <- somaCusto / qtdEntregas

      exibirLinha

      escreval("Quantidade total de entregas ...: ", qtdEntregas)
      escreval("Aguardando .....................: ", contStatus[AGUARDANDO])
      escreval("Em transporte ..................: ", contStatus[EM_TRANSPORTE])
      escreval("Entregues ......................: ", contStatus[ENTREGUE])
      escreval("Urgentes .......................: ", contPrio[URGENTE])
      escreval("Prioritarias ...................: ", contPrio[PRIORITARIA])
      escreval("Normais ........................: ", contPrio[NORMAL])

      aguardarEnter(">> Digite 1 e aperte ENTER para continuar...")

      escreval("Peso total transportado ........: ", somaPeso:10:2, " kg")
      escreval("Media de peso ..................: ", mediaPeso:10:2, " kg")
      escreval("Valor total das mercadorias ....: R$ ", somaValor:10:2)
      escreval("Valor medio das mercadorias ....: R$ ", mediaValor:10:2)
      escreval("Custo total estimado ...........: R$ ", somaCusto:10:2)
      escreval("Custo medio das entregas .......: R$ ", mediaCusto:10:2)

      aguardarEnter(">> Digite 1 e aperte ENTER para continuar...")

      escreval("Entrega de maior valor .........: cod ", vCodigo[idxMaiorValor], " (", vDestinatario[idxMaiorValor], ") - R$ ", vValor[idxMaiorValor]:10:2)
      escreval("Entrega de maior peso ..........: cod ", vCodigo[idxMaiorPeso], " (", vDestinatario[idxMaiorPeso], ") - ", vPeso[idxMaiorPeso]:8:2, " kg")
      escreval("Entrega de maior distancia .....: cod ", vCodigo[idxMaiorDist], " (", vDestinatario[idxMaiorDist], ") - ", vDistancia[idxMaiorDist], " km")

      aguardarEnter(">> Digite 1 e aperte ENTER para continuar...")

      maiorQtdRegiao <- 0

      escreval("Entregas recebidas por regiao:")

      para j de 1 ate MAX_REGIOES faca
         escreval("   ", nomeRegiao[j], ": ", contRegiao[j])

         se contRegiao[j] > maiorQtdRegiao entao
            maiorQtdRegiao <- contRegiao[j]
         fimse
      fimpara

      escreval("Regiao que mais recebeu entregas (", maiorQtdRegiao, " entrega(s)): ")

      para j de 1 ate MAX_REGIOES faca
         se contRegiao[j] = maiorQtdRegiao entao
            escreval("   -> ", nomeRegiao[j])
         fimse
      fimpara

      exibirLinha
   fimse
fimprocedimento

funcao distanciaTotal(posicao: inteiro): inteiro
var
   resultado: inteiro
inicio
   se posicao >= tamPercurso entao
      resultado <- 0
   senao
      resultado <- obterDistancia(percurso[posicao], percurso[posicao + 1]) + distanciaTotal(posicao + 1)
   fimse

   retorne resultado

procedimento calcularPercurso
var
   k, regiaoLida: inteiro
   regiaoOk: logico
inicio
   escreval("=== CALCULAR PERCURSO RECURSIVAMENTE ===")
   exibirRegioes

   lerInteiroNoIntervalo("Quantidade de regioes do percurso", 2, MAX_PERCURSO, tamPercurso)

   para k de 1 ate tamPercurso faca
      repita
         regiaoOk <- verdadeiro

         escreval("Regiao numero ", k, " do percurso:")
         lerInteiroNoIntervalo("Codigo da regiao", 1, MAX_REGIOES, regiaoLida)

         se k > 1 entao
            se regiaoLida = percurso[k - 1] entao
               escreval("ERRO: a regiao nao pode ser igual a anterior do percurso.")
               regiaoOk <- falso
            fimse
         fimse
      ate regiaoOk

      percurso[k] <- regiaoLida
   fimpara

   escreva("Percurso: ")

   para k de 1 ate tamPercurso faca
      escreva(nomeRegiao[percurso[k]])

      se k < tamPercurso entao
         escreva(" -> ")
      fimse
   fimpara

   escreval("")
   escreval("Distancia total do percurso: ", distanciaTotal(1), " km")
fimprocedimento
